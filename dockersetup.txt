# Build the Docker image
docker build -t gspc-feynman .

# Run the container locally (for testing)
docker run -it --hostname waas --cap-add=SYS_ADMIN gspc-feynman
docker run -it --hostname $(hostname) --cap-add=SYS_ADMIN gspc-feynman

# Tag the image for Singular-GPISpace organization
docker tag gspc-feynman ghcr.io/singular-gpispace/gspc-feynman:latest

# Login to GitHub Container Registry
# Replace YOUR_TOKEN with your GitHub Personal Access Token
echo YOUR_TOKEN | docker login ghcr.io -u $(whoami) --password-stdin

# Push the image to Singular-GPISpace organization
docker push ghcr.io/singular-gpispace/gspc-feynman:latest
